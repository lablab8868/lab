4)MEMORY ALLOCATION ON HEAP AND STACK
> nano memory.c

#include <stdio.h>
#include <stdlib.h>

int main()
{
    int stack_var = 10;
    int *heap_var;

    heap_var = (int *)malloc(sizeof(int));

    if (heap_var == NULL)
    {
        printf("Memory allocation failed\n");
        return 1;
    }

    *heap_var = 20;

    printf("Stack variable value : %d\n", stack_var);
    printf("Stack variable address : %p\n",
           (void *)&stack_var);

    printf("Heap variable value : %d\n", *heap_var);
    printf("Heap variable address : %p\n",
           (void *)heap_var);

    free(heap_var);

    return 0;
}
 
> gcc -Wall -Wextra -g memory.c -o memory

#!/usr/bin/tclsh

# If-else control flow
set num 7

if {$num % 2 == 0} {
    puts "$num is even"
} else {
    puts "$num is odd"
}

# For loop
puts "Numbers 1 to 5:"
for {set i 1} {$i <= 5} {incr i} {
    puts $i
}

# Tcl list
set fruits [list Apple Banana Mango]
lappend fruits Orange

puts "Fruit list:"
foreach f $fruits {
    puts $f
}

# Tcl associative array
set marks(Ravi) 85
set marks(Kiran) 90
set marks(Asha) 78

puts "Student marks:"
foreach name [lsort [array names marks]] {
    puts "$name -> $marks($name)"
}

# Add entry text to the listbox
proc addItem {} {
    set val [.ent get]

    if {$val ne ""} {
        .lst insert end $val
        .ent delete 0 end
    }
}

button .addBtn -text "Add Item" -command addItem
pack .addBtn -pady 5

# Set initial keyboard focus
focus .ent

# Modal confirmation dialog
proc showConfirm {} {
    if {[winfo exists .confirm]} {
        return
    }

    toplevel .confirm
    wm title .confirm "Confirmation"

    label .confirm.msg -text "Item added successfully!"
    button .confirm.ok -text "OK" -command {
        grab release .confirm
        destroy .confirm
    }

    pack .confirm.msg .confirm.ok -padx 10 -pady 10

    grab .confirm
    focus .confirm.ok
}

button .confirmBtn -text "Show Confirmation" \
    -command showConfirm

    echo "Both numbers are positive."
else
    echo "At least one number is not positive."
fi

if [ "$a" -gt 0 ] || [ "$b" -gt 0 ]; then
    echo "At least one number is positive."
else

    echo "The first number is not positive."
fi

#!/bin/bash

echo "Enter marks (0-100):"
read marks

if [ "$marks" -ge 90 ] && [ "$marks" -le 100 ]; then
    echo "Grade: A"
elif [ "$marks" -ge 75 ] && [ "$marks" -lt 90 ]; then
    echo "Grade: B"
elif [ "$marks" -ge 60 ] && [ "$marks" -lt 75 ]; then
    echo "Grade: C"
elif [ "$marks" -ge 0 ] && [ "$marks" -lt 60 ]; then
    echo "Grade: F"
else
    echo "Invalid marks."
fi

7(l) Metacharacters Using a Perl Script

nano metacharacters.pl


#!/usr/bin/perl
use strict;
use warnings;

my $text = "12345 2026-10-14 Laptop A123";

if ($text =~ /^(\d+)/) {
    print "Order number: $1\n";
}

if ($text =~ /(\d{4}-\d{2}-\d{2})/) {
    print "Date: $1\n";
}

if ($text =~ /([A-Za-z]+)\s+([A-Za-z]\d+)$/) {
    print "Item name: $1\n";
    print "Item code: $2\n";
}
>chmod +x metacharacters.pl
