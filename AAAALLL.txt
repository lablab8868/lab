4)MEMORY ALLOCATION ON HEAP AND STACK
> nano memory.c

#include <stdio.h>
#include <stdlib.h>

int main()
{
    int stack_var = 10;
    int *heap_var;

    heap_var = (int *)malloc(sizeof(int));

    if (heap_var == NULL)
    {
        printf("Memory allocation failed\n");
        return 1;
    }

    *heap_var = 20;

    printf("Stack variable value : %d\n", stack_var);
    printf("Stack variable address : %p\n",
           (void *)&stack_var);

    printf("Heap variable value : %d\n", *heap_var);
    printf("Heap variable address : %p\n",
           (void *)heap_var);

    free(heap_var);

    return 0;
}

nohup ./backup.sh > backup.log 2>&1 &

The process ID printed by echo $! is the PID. The log is saved to backup.log because output was explicitly redirected.
The standard nohup behaviour, when output is not redirected and the terminal is unavailable, is to append output to

kill PID

EXERCISE 8: TCL CONTROL FLOW COMMANDS AND DATA

Install Tcl

Create Tcl program
Create the file using nano:
nano control_ds.tcl
#!/usr/bin/tclsh
# if-else
set num 7
if {$num % 2 == 0} {
 puts "$num is even"
} else {
 puts "$num is odd"
}
# for loop
puts "Numbers 1 to 5:"
for {set i 1} {$i <= 5} {incr i} {
 puts " $i"
}
# Tcl list
set fruits [list Apple Banana Mango]
lappend fruits Orange
puts "Fruit list:"
foreach f $fruits {
 puts " $f"
}
# Associative array
set marks(Ravi) 85
set marks(Kiran) 90
set marks(Asha) 78
puts "Student marks:"
foreach name [array names marks] {
 puts " $name -> $marks($name)"
}

If both variables are empty or wish reports “couldn't connect to display”, configure/update WSL GUI support before

Create the Tk script
nano widgets_demo.tcl
#!/usr/bin/wish
# Entry widget
label .lbl -text "Enter an item:"
entry .ent -width 25
pack .lbl -pady 5
pack .ent -pady 5
# Listbox widget
listbox .lst -height 6 -width 25
.lst insert end Pen Book Bag
pack .lst -pady 5
# Add entry text to listbox
proc addItem {} {
 set val [.ent get]
 if {$val ne ""} {
 .lst insert end $val
 .ent delete 0 end
 }
}
button .addBtn -text "Add Item" -command addItem
pack .addBtn -pady 5
# Initial keyboard focus
focus .ent
# Modal confirmation dialog
proc showConfirm {} {
 if {[winfo exists .confirm]} {
 return
 }
 toplevel .confirm
 wm title .confirm "Confirmation"
 label .confirm.msg -text "Item added successfully!"
 button .confirm.ok -text "OK" -command {
 grab release .confirm
 destroy .confirm
 }
 pack .confirm.msg .confirm.ok -padx 10 -pady 10
 grab .confirm
 focus .confirm.ok
}
button .confirmBtn -text "Show Confirmation" -command showConfirm
pack .confirmBtn -pady 5
Run the Tk program
wish widgets_demo.tcl
A window should appear with a label, Entry box, Listbox containing Pen, Book and Bag, an Add Item button and a Show
Confirmation button. Type Eraser in the Entry box and click Add Item. Click Show Confirmation to open the modal dialog;
click OK to close it.
