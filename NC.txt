import java.util.Scanner;

class CaesarCipher {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        System.out.print("Enter text: ");
        String text = sc.nextLine().toUpperCase();

        System.out.print("Enter key: ");
        int key = sc.nextInt();

        String cipher = "";

        for (int i = 0; i < text.length(); i++) {
            char ch = text.charAt(i);

            if (ch >= 'A' && ch <= 'Z') {
                cipher += (char) ((ch - 'A' + key) % 26 + 'A');
            } else {
                cipher += ch;
            }
        }

        System.out.println("Encrypted: " + cipher);

        String plain = "";

        for (int i = 0; i < cipher.length(); i++) {
            char ch = cipher.charAt(i);

            if (ch >= 'A' && ch <= 'Z') {
                plain += (char) ((ch - 'A' - key + 26) % 26 + 'A');
            } else {
                plain += ch;
            }
        }

        System.out.println("Decrypted: " + plain);
        sc.close();
    }
}

import java.util.Scanner;

class PlayfairCipher {

    static char[][] a = new char[5][5];

    static String process(String text, int dir) {
        String result = "";

        for (int i = 0; i < text.length(); i += 2) {
            int r1 = 0, c1 = 0, r2 = 0, c2 = 0;

            for (int r = 0; r < 5; r++)
                for (int c = 0; c < 5; c++) {
                    if (a[r][c] == text.charAt(i)) {
                        r1 = r; c1 = c;
                    }
                    if (a[r][c] == text.charAt(i + 1)) {
                        r2 = r; c2 = c;
                    }
                }

            if (r1 == r2) {
                result += a[r1][(c1 + dir + 5) % 5];
                result += a[r2][(c2 + dir + 5) % 5];
            }
            else if (c1 == c2) {
                result += a[(r1 + dir + 5) % 5][c1];
                result += a[(r2 + dir + 5) % 5][c2];
            }
            else {
                result += a[r1][c2];
                result += a[r2][c1];
            }
        }
        return result;
    }

    public static void main(String[] args) {

        Scanner sc = new Scanner(System.in);

        System.out.print("Enter text: ");
        String text = sc.nextLine().toUpperCase()
                        .replace("J", "I")
                        .replaceAll("[^A-Z]", "");

        System.out.print("Enter key: ");
        String key = sc.nextLine().toUpperCase()
                       .replace("J", "I")
                       .replaceAll("[^A-Z]", "");

        String s = "";

        for (char c : (key + "ABCDEFGHIKLMNOPQRSTUVWXYZ").toCharArray())
            if (s.indexOf(c) == -1)
                s += c;


        for (int i = 0; i < 25; i++)
            a[i / 5][i % 5] = s.charAt(i);

        String p = "";

        for (int i = 0; i < text.length(); i++) {
            p += text.charAt(i);

            if (i + 1 < text.length() &&
                text.charAt(i) == text.charAt(i + 1))
                p += 'X';
        }

        if (p.length() % 2 != 0)
            p += 'X';

        String cipher = process(p, 1);
        String decrypted = process(cipher, -1);

        System.out.println("Encrypted: " + cipher);
        System.out.println("Decrypted: " + decrypted);

        sc.close();
    }

import java.util.Scanner;

class HillCipher {

    static int modInverse(int a) {
        for (int i = 1; i < 26; i++)
            if ((a * i) % 26 == 1)
                return i;
        return -1;
    }

    static int mod(int n) {
        return (n % 26 + 26) % 26;
    }

    static String process(String text, int[][] k) {
        String result = "";

        for (int i = 0; i < text.length(); i += 3) {
            for (int r = 0; r < 3; r++) {
                int sum = 0;

                for (int c = 0; c < 3; c++)
                    sum += k[r][c] * (text.charAt(i + c) - 'A');

                result += (char) ('A' + mod(sum));
            }
        }
        return result;
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int[][] k = new int[3][3];

        System.out.println("Enter 3x3 key matrix:");

        for (int i = 0; i < 3; i++)
            for (int j = 0; j < 3; j++)
                k[i][j] = sc.nextInt();

        sc.nextLine();

        System.out.print("Enter plain text: ");
        String text = sc.nextLine().toUpperCase()
                .replaceAll("[^A-Z]", "");

        while (text.length() % 3 != 0)
            text += "X";

        int a = k[0][0], b = k[0][1], c = k[0][2];
        int d = k[1][0], e = k[1][1], f = k[1][2];
        int g = k[2][0], h = k[2][1], i = k[2][2];

        int det = mod(a*(e*i-f*h) - b*(d*i-f*g)
                      + c*(d*h-e*g));

        int di = modInverse(det);

        if (di == -1) {
            System.out.println("Invalid key matrix!");
            sc.close();
            return;
        }

        int[][] inv = {
            {mod((e*i-f*h)*di), mod((c*h-b*i)*di),
             mod((b*f-c*e)*di)},
            {mod((f*g-d*i)*di), mod((a*i-c*g)*di),
             mod((c*d-a*f)*di)},
            {mod((d*h-e*g)*di), mod((b*g-a*h)*di),
             mod((a*e-b*d)*di)}
        };

        String cipher = process(text, k);

        System.out.println("Encrypted text: " + cipher);
        System.out.println("Decrypted text: " + process(cipher, inv));

        sc.close();
    }
}

Experiment 1(d) – Vigenère Cipher

File: VigenereCipher.java

Simple idea: each letter is shifted by a repeating keyword.

PROGRAM


import java.util.Scanner;

class VigenereCipher {

    public static void main(String[] args) {

        Scanner sc = new Scanner(System.in);

        System.out.print("Enter text: ");
        String text = sc.nextLine().toUpperCase();

        System.out.print("Enter key: ");
        String key = sc.nextLine().toUpperCase();

        String enc = "";
        String dec = "";

        for (int i = 0; i < text.length(); i++) {

            int a = text.charAt(i) - 'A';
            int b = key.charAt(i % key.length()) - 'A';

            enc = enc + (char)((a + b) % 26 + 'A');
        }

        for (int i = 0; i < enc.length(); i++) {

            int a = enc.charAt(i) - 'A';
            int b = key.charAt(i % key.length()) - 'A';

            dec = dec + (char)((a - b + 26) % 26 + 'A');
        }

        System.out.println("Encrypted: " + enc);
        System.out.println("Decrypted: " + dec);

        sc.close();
    }
}

Experiment 2(a) – Rail Fence Cipher

File: RailFence.java

Simple idea: write letters diagonally across rows, then read row by row.

PROGRAM


import java.util.Scanner;

class RailFence {

    public static void main(String[] args) {

        Scanner sc = new Scanner(System.in);

        System.out.print("Enter text: ");
        String text = sc.nextLine().toUpperCase();

        System.out.print("Enter key: ");
        int key = sc.nextInt();

        String enc = "";

        // Encryption
        for (int i = 0; i < key; i++) {
            for (int j = i; j < text.length(); j += key) {
                enc = enc + text.charAt(j);
            }
        }

        System.out.println("Encrypted: " + enc);

        // Decryption
        String dec = "";
        int n = text.length();
        int index = 0;

        char rail[][] = new char[key][n];

        for (int i = 0; i < key; i++) {
            for (int j = i; j < n; j += key) {
                rail[i][j] = enc.charAt(index++);
            }
        }

        for (int j = 0; j < n; j++) {
            for (int i = 0; i < key; i++) {
                if (rail[i][j] != '\0') {
                    dec = dec + rail[i][j];
                }
            }
        }

        System.out.println("Decrypted: " + dec);

        sc.close();
    }
}

Experiment 2(b) – Row and Column Transposition

File: RowColumnTransposition.java

Simple idea: write text in rows and read columns in the alphabetical order of the keyword. X is padding.

PROGRAM


import java.util.Scanner;

class RowColumn {
    public static void main(String[] args) {

        Scanner sc = new Scanner(System.in);

        System.out.print("Enter text: ");
        String text = sc.nextLine().toUpperCase();

        System.out.print("Enter key: ");
        String key = sc.nextLine();

        int cols = key.length();
        int rows = (int)Math.ceil((double)text.length() / cols);

        char a[][] = new char[rows][cols];
        int k = 0;

        // Fill matrix
        for (int i = 0; i < rows; i++)
            for (int j = 0; j < cols; j++)
                a[i][j] = (k < text.length()) ? text.charAt(k++) : 'X';

        // Encryption
        String enc = "";

        for (int n = 1; n <= cols; n++)
            for (int j = 0; j < cols; j++)
                if (key.charAt(j) - '0' == n)
                    for (int i = 0; i < rows; i++)
                        enc += a[i][j];

        System.out.println("Encrypted: " + enc);

        // Decryption
        k = 0;

        for (int n = 1; n <= cols; n++)
            for (int j = 0; j < cols; j++)
                if (key.charAt(j) - '0' == n)
                    for (int i = 0; i < rows; i++)
                        a[i][j] = enc.charAt(k++);

        String dec = "";

        for (int i = 0; i < rows; i++)
            for (int j = 0; j < cols; j++)
                dec += a[i][j];

        System.out.println("Decrypted: " + dec);

        sc.close();
    }
}

Experiment 3 – DES Algorithm

File: DESAlgorithm.java

Simple idea: DES uses a secret key to encrypt and decrypt data. DES is outdated and should not be used for real security.

PROGRAM

import javax.crypto.Cipher;
import javax.crypto.spec.SecretKeySpec;
import java.util.*;

class DES {

    public static void main(String[] args) throws Exception {

        Scanner sc = new Scanner(System.in);

        System.out.print("Enter plaintext: ");
        String text = sc.nextLine();

        System.out.print("Enter 8 character key: ");
        String key = sc.nextLine();

        SecretKeySpec k = new SecretKeySpec(key.getBytes(), "DES");

        Cipher c = Cipher.getInstance("DES");

        // Encryption
        c.init(Cipher.ENCRYPT_MODE, k);
        byte[] encrypted = c.doFinal(text.getBytes());

        // Decryption
        c.init(Cipher.DECRYPT_MODE, k);
        byte[] decrypted = c.doFinal(encrypted);

        System.out.println("Encrypted Text: " +
                Base64.getEncoder().encodeToString(encrypted));

        System.out.println("Decrypted Text: " +
                new String(decrypted));

        sc.close();
    }
}

Experiment 5 – RSA Algorithm (Java)

File: RSAAlgorithm.java

Simple idea: RSA uses a public key to encrypt and a private key to decrypt. This small-number version is for lab learning only.

PROGRAM

import java.util.Scanner;

class RSAAlgorithm {

    static long pow(long b, long e, long m) {
        long r = 1;

        while (e > 0) {
            if ((e & 1) == 1)
                r = r * b % m;

            b = b * b % m;
            e >>= 1;
        }

        return r;
    }

    static long gcd(long a, long b) {
        return b == 0 ? a : gcd(b, a % b);
    }

    public static void main(String[] args) {

        Scanner sc = new Scanner(System.in);

        System.out.print("Enter prime p: ");
        long p = sc.nextLong();

        System.out.print("Enter prime q: ");
        long q = sc.nextLong();

        System.out.print("Enter message number (< n): ");
        long m = sc.nextLong();

        long n = p * q;
        long phi = (p - 1) * (q - 1);
        long e = 3;

        while (gcd(e, phi) != 1)
            e += 2;

        long d = 1;

        while ((d * e) % phi != 1)
            d++;

        long c = pow(m, e, n);
        long dec = pow(c, d, n);

        System.out.println("n: " + n);
        System.out.println("phi: " + phi);

        System.out.println("Public key: (" + e + ", " + n + ")");
        System.out.println("Private key: (" + d + ", " + n + ")");

        System.out.println("Encrypted: " + c);
        System.out.println("Decrypted: " + dec);

        sc.close();
    }
}

Experiment 6 – Diffie-Hellman Key Exchange

File: DiffieHellman.java

Simple idea: two users calculate the same secret key without sending the secret key directly.

PROGRAM

import java.util.Scanner;

import java.util.Scanner;

class DiffieHellman {

    public static void main(String[] args) {

        Scanner sc = new Scanner(System.in);

        System.out.print("Enter prime number: ");
        int p = sc.nextInt();

        System.out.print("Enter primitive root: ");
        int g = sc.nextInt();

        System.out.print("Enter private key of A: ");
        int a = sc.nextInt();

        System.out.print("Enter private key of B: ");
        int b = sc.nextInt();

        int A = 1;
        for (int i = 1; i <= a; i++)
            A = (A * g) % p;

        int B = 1;
        for (int i = 1; i <= b; i++)
            B = (B * g) % p;

        int keyA = 1;
        for (int i = 1; i <= a; i++)
            keyA = (keyA * B) % p;

        int keyB = 1;
        for (int i = 1; i <= b; i++)
            keyB = (keyB * A) % p;

        System.out.println("Public key of A: " + A);
        System.out.println("Public key of B: " + B);

        System.out.println("Shared key of A: " + keyA);
        System.out.println("Shared key of B: " + keyB);

        sc.close();
    }
}

Experiment 7 – Digital Signature (DSA demo)

File: DigitalSignature.java

Simple idea: a digital signature helps verify who signed a message and whether it changed. The signature string changes each run. This is a simple DSA demonstration, not the exact DSS implementation in the PDF.

PROGRAM

import java.util.*;
import java.security.*;

class DigitalSignature {

    public static void main(String[] args) throws Exception {

        Scanner sc = new Scanner(System.in);

        System.out.print("Enter message: ");
        String message = sc.nextLine();

        System.out.print("Enter message for verification: ");
        String verifyMessage = sc.nextLine();

        // Generate keys
        KeyPairGenerator gen = KeyPairGenerator.getInstance("RSA");
        gen.initialize(2048);

        KeyPair pair = gen.generateKeyPair();

        PrivateKey privateKey = pair.getPrivate();
        PublicKey publicKey = pair.getPublic();

        // Create signature
        Signature sign = Signature.getInstance("SHA256withRSA");

        sign.initSign(privateKey);
        sign.update(message.getBytes());

        byte[] signature = sign.sign();

        // Verify signature
        sign.initVerify(publicKey);
        sign.update(verifyMessage.getBytes());

        boolean result = sign.verify(signature);

        System.out.println("Digital Signature: " +
                Base64.getEncoder().encodeToString(signature));

        System.out.println("Signature Verified: " + result);

        sc.close();
    }
}
